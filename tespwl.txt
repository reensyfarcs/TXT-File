    <h2>Switch Case Nilai</h2>
    <form method="post">
        Masukkan Nilai (0-100): <input type="number" name="nilai_switch" required>
        <input type="submit" name="submit_switch" value="OK">
    </form>

    <?php
    if (isset($_POST['submit_switch'])) {
        $nilai_switch = $_POST['nilai_switch'];
        switch (true) {
            case ($nilai_switch >= 85 && $nilai_switch <= 100): $grade = "A"; break;
            case ($nilai_switch >= 70): $grade = "B"; break;
            case ($nilai_switch >= 60): $grade = "C"; break;
            case ($nilai_switch >= 50): $grade = "D"; break;
            default: $grade = "E";
        }
        echo "<h3>Hasil: Nilai $nilai_switch = $grade</h3>";
    }
